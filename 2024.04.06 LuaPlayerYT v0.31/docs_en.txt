==============
Buttons/Кнопки
==============

buttons.read() - read buttons (usage pad = buttons.read(); oldpad = pad)

buttons.Lx() - read stick on x axis

buttons.Ly() - read stick on y axis

buttons.pressed[button](pad/oldpad) - check for just pressed buttons

buttons.hold[button](pad/oldpad) - check for hold buttons

buttons.released[button](pad/oldpad) - check for just released buttons

button variables:
	- start
	- select
	- up
	- down
	- left
	- right
	- triangle
	- cross
	- square
	- circle
	- l
	- r

==========
Color/Цвет
==========

Color.new(r, g, b, [a]) - create new color
	- r - red (0-255)
	- g - green (0-255)
	- b - blue (0-255)
	- a - alpha (0-255) [optional, default - 255]

Color.get(color, [R/G/B/A]) - get specific value
	- color - color
	- R/G/B/A - specific channel [optional, default - {r, g, b, a}]

===============
Graphix/Графика
===============

screen.clear([color]) - clear the screen
	- color - color [optional, default - black]

scree.flip() - swap the buffers

screen.drawRect(x, y, w, h, color, [alpha]) - draw a rectangle

screen.drawCircle(x, y, raduis, color) - draw a circle

screen.drawTriangle(x1, y1, x2, y2, x3, y3, color) - draw a triangle

Image.load(path) - load an image

Image.unload(img) - unload an image
	- img - image variable
	[UNLOAD REQUIRED WHEN IMAGE IS USELESS ANYMORE]

Image.w(img) - get an image width

Image.h(img) - get an image height

Image.draw(img, x, y, [w], [h], [color], [srcx], [srcy], [srcw], [srch], [r], [a], [alMode]) - draw an image
	- c - color to apply [optional, default - nothing]
	
	- srcx, srcy - start crop coordinates [optional, default - 0, 0]
	- srcw, srch - end crop coordinates [optional, default - Image.w(), Image.h()]
	
	- alMode - align mode [optional, default - Image.lUP]

Image.draweasy(img, x, y, [c], [r], [a], [alMode]) - draw an image easy
	- c - color to apply [optional, default - nothing]

	- alMode - align mode [optional, default - Image.lUP]

alMode variables:
	- Image.lUp - left upper
	- Image.Center - center
	- Image.rUp - right upper
	- Image.lDn - left down
	- Image.rDn - right down

===============
intraFont/Шрифт
===============

intraFont.load(path) - load a .pgf font

intraFont.setStyle(font, s, c, sc, r, [align]) - set font style
	- s - font size
	- c - font color
	- sc - shadow color (Color.new(0,0,0,0) recommended)
	- r - rotate angle
	- align - align mode

	align modes:
		- intraFont.ALIGN_LEFT
		- intraFont.ALIGN_CENTER
		- intraFont.ALIGH_RIGHT

intraFont.print(font, x, y, text) - print a text

intraFont.printColumn(font, x, y, w, c, s) - print text as column
	- w - width in pixels
	
	- c - text color
	- s - text size

intraFont.printUnderlined(font, x, y, text, c, c2, [s])
	- c - text color
	- c2 - underline color
	- s - text size
    
intraFont.printContoured(font, x, y, text, c, c2, [s], [r])
	- c - text color
	- c2 - contour color
	- s - text size
	- r - text rotation

intraFont.printBackground(font, x, y, text, c, c2, [s])
	- c - text color
	- c2 - background color
	- s - text size

intraFont.printStriked(font, x, y, text, c, c2, [s])
	- c - text color
	- c2 - strike color
	- s - text size

intraFont.printShadowed(font, x, y, text, c, c2, SHr, lD, [s], [r])
	- c - text color
	- c2 - shadow color
	- SHr - light angle
	- lD - light distance
	- s - text size
	- r - text rotation

   
intraFont.printReversed(font, x, y, text, c, [s], [r])
	- c - text color
	- s - text size
	- r - text rotation

intraFont.printColored(font, x, y, text, c, [s], [r])
	- c - text color
	- s - text size
	- r - text rotation


intraFont.size(font, s) - change font size
	- s - new size

intraFont.textW(font, text) - get string lenght in pixels

intraFont.textH(font) - get font height in pixels

==========
Sound/Звук
==========

NOTE: sound files should have frequency not more than 44100

sound.play(path, channel, loadToRAM, loop) - play file
	- channel - channel to load
	- loadToRAM - load to RAM (true for continue playing wav after turn on from sleep mode, but RAM will become less slowly)
	
	PSP GO will continue playing anyway;
	
	(.ogg/.at3 don't support that function)

sound.info() - read ID3 tags of mp3 file
	return table:
		- title
		- artist
		- album
		- year
		- genre
		- id3v
		- track

sound.pause(channel) - pause playing

sound.volume(channel, volume) - change volume

sound.stop(channel) - stop playing

sound.state(channel) - get channel info
	return:
		- state (playing/paused/stopped)
		- free (true/false)
		- volume (0-100)

sound.unload(channel) - unload from channel (only for .wav/.at3/.ogg, for .mp3 enough sound.stop())
	channels:
		==============
		- sound.MP3
		==============
		- sound.WAV_1
		...
		- sound.WAV_31
		==============
		- sound.OGG_1
		...
		- sound.OGG_10
		==============
		- sound.AT3_1
		- sound.AT3_2
		==============
